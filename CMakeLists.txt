cmake_minimum_required(VERSION 3.28)
project(BenScrSpatial VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(BENSCR_SPATIAL_BUILD_TESTS "Build contract tests" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_SPATIAL_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_SPATIAL_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(benscr_spatial STATIC src/point_index.cpp)
add_library(BenScr::Spatial ALIAS benscr_spatial)
set_target_properties(benscr_spatial PROPERTIES EXPORT_NAME Spatial CXX_EXTENSIONS OFF)
target_compile_features(benscr_spatial PUBLIC cxx_std_20)
target_include_directories(benscr_spatial PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(benscr_spatial_checks target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BENSCR_SPATIAL_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported POSIX host.")
        endif()
    endif()
endfunction()
benscr_spatial_checks(benscr_spatial)

if(BUILD_TESTING AND BENSCR_SPATIAL_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(spatial_contract_tests tests/spatial_tests.cpp)
    target_link_libraries(spatial_contract_tests PRIVATE BenScr::Spatial Threads::Threads)
    benscr_spatial_checks(spatial_contract_tests)
    add_test(NAME spatial_contract COMMAND spatial_contract_tests)
    add_executable(spatial_header_test tests/header.cpp)
    target_link_libraries(spatial_header_test PRIVATE BenScr::Spatial)
    benscr_spatial_checks(spatial_header_test)
    add_test(NAME standalone_header COMMAND spatial_header_test)
    add_executable(spatial_allocation_tests tests/allocation_tests.cpp)
    target_link_libraries(spatial_allocation_tests PRIVATE BenScr::Spatial)
    benscr_spatial_checks(spatial_allocation_tests)
    add_test(NAME allocation_rollback COMMAND spatial_allocation_tests)
endif()
if(BENSCR_SPATIAL_BUILD_EXAMPLES)
    add_executable(spatial_example examples/spatial.cpp)
    target_link_libraries(spatial_example PRIVATE BenScr::Spatial)
    benscr_spatial_checks(spatial_example)
endif()

install(TARGETS benscr_spatial EXPORT BenScrSpatialTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/BenScrSpatial)
configure_package_config_file(cmake/BenScrSpatialConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrSpatialConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrSpatial)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BenScrSpatialConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BenScrSpatialConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrSpatialConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrSpatial)
install(EXPORT BenScrSpatialTargets NAMESPACE BenScr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrSpatial)
